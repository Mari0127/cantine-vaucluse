function doPost(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("CANTINE GIONO"); var params = JSON.parse(e.postData.contents); // On cherche si la période (Lundi, Sept, etc.) existe déjà var data = sheet.getDataRange().getValues(); var rowToUpdate = -1; for (var i = 1; i < data.length; i++) { if (data[i][0] == params.periode) { rowToUpdate = i + 1; // +1 car les lignes Sheets commencent à 1 break; } } // Si elle existe, on met à jour, sinon on crée une nouvelle ligne if (rowToUpdate != -1) { sheet.getRange(rowToUpdate, 2, 1, 9).setValues([[ params.vRouge, params.vBlanche, params.veget, params.poisson, params.pain, params.fruits, params.emb, params.dechets, params.serv ]]); } else { sheet.appendRow([ params.periode, params.vRouge, params.vBlanche, params.veget, params.poisson, params.pain, params.fruits, params.emb, params.dechets, params.serv ]); } return ContentService.createTextOutput(JSON.stringify({"status":"success"})).setMimeType(ContentService.MimeType.JSON); }
